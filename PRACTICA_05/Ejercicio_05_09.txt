#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

int calcularFactorial(int numero);

int main()
{
    int numero = 0, resultado = 0;

    srand(time(0));

    numero = rand() % 10 + 1;
    resultado = calcularFactorial(numero);

    cout << "Numero generado: " << numero << endl;
    cout << "Factorial: " << resultado << endl;

    return 0;
}

int calcularFactorial(int numero)
{
    int factorial = 1;

    for (int i = 1; i <= numero; i++)
    {
        factorial = factorial * i;
    }

    return factorial;
}
