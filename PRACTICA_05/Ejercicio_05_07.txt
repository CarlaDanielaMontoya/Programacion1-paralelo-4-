#include <iostream>
using namespace std;

void agregarNota(double &sumaTotal, int &cantidadNotas,
                 double nuevaNota);

int main()
{
    int n = 0, cantidadNotas = 0;
    double sumaTotal = 0, nuevaNota = 0;

    do
    {
        cout << "Ingrese la cantidad de notas: ";
        cin >> n;
    } while (n <= 0);

    for (int i = 1; i <= n; i++)
    {
        cout << "Ingrese la nota " << i << ": ";
        cin >> nuevaNota;

        agregarNota(sumaTotal, cantidadNotas, nuevaNota);
    }

    cout << "Suma de las notas: " << sumaTotal << endl;
    cout << "Cantidad de notas: " << cantidadNotas << endl;
    cout << "Promedio: " << sumaTotal / cantidadNotas << endl;

    return 0;
}

void agregarNota(double &sumaTotal, int &cantidadNotas,
                 double nuevaNota)
{
    sumaTotal = sumaTotal + nuevaNota;
    cantidadNotas++;
}
