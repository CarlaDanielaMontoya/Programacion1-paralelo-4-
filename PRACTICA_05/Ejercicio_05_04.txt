#include <iostream>
using namespace std;

double CalcularPrecioTotal(double precio, double impuesto = 13);

int main()
{
    double precio = 0, impuesto = 0;

    do
    {
        cout << "Ingrese el precio base: ";
        cin >> precio;
    } while (precio < 0);

    cout << "Total con impuesto del 13%: ";
    cout << CalcularPrecioTotal(precio) << endl;

    do
    {
        cout << "Ingrese otro porcentaje de impuesto: ";
        cin >> impuesto;
    } while (impuesto < 0);

    cout << "Total con el impuesto ingresado: ";
    cout << CalcularPrecioTotal(precio, impuesto) << endl;

    return 0;
}

double CalcularPrecioTotal(double precio, double impuesto)
{
    double total = 0;

    total = precio + (precio * impuesto / 100);

    return total;
}
