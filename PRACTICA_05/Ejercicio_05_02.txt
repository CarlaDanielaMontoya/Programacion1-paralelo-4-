#include <iostream>
using namespace std;

void ModificarValores(int a, int &b);

int main()
{
    int a = 0, b = 0;

    cout << "Ingrese el primer numero: ";
    cin >> a;

    cout << "Ingrese el segundo numero: ";
    cin >> b;

    cout << "Antes de llamar a la funcion:" << endl;
    cout << "a = " << a << endl;
    cout << "b = " << b << endl;

    ModificarValores(a, b);

    cout << "Despues de llamar a la funcion:" << endl;
    cout << "a = " << a << endl;
    cout << "b = " << b << endl;

    return 0;
}

void ModificarValores(int a, int &b)
{
    a = a * 2;
    b = b + 10;

    cout << "Dentro de la funcion:" << endl;
    cout << "a = " << a << endl;
    cout << "b = " << b << endl;
}
