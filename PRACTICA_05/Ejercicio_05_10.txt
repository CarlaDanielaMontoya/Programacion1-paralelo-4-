#include <iostream>
#include <cstdlib>
#include <ctime>
using namespace std;

bool esPrimo(int numero);

int main()
{
    int n = 0, numero = 0, cantidadPrimos = 0;

    srand(time(0));

    do
    {
        cout << "Ingrese la cantidad de numeros: ";
        cin >> n;
    } while (n <= 0);

    for (int i = 1; i <= n; i++)
    {
        numero = rand() % 10000 + 1;

        cout << "Numero " << i << ": " << numero << endl;

        if (esPrimo(numero))
        {
            cantidadPrimos++;
        }
    }

    cout << "Cantidad de numeros primos: " << cantidadPrimos << endl;

    return 0;
}

bool esPrimo(int numero)
{
    int divisores = 0;

    for (int i = 1; i <= numero; i++)
    {
        if (numero % i == 0)
        {
            divisores++;
        }
    }

    if (divisores == 2)
    {
        return true;
    }
    else
    {
        return false;
    }
}
